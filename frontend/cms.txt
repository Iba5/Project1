/**
 * CMS abstraction layer
 * ====================
 *
 * This folder is the single integration point for content. The rest of the
 * app imports data ONLY from here:
 *
 *     import { getProducts, getIndustries, getGalleryItems } from "@/lib/cms";
 *
 * Two adapters exist:
 *   - static-adapter.ts  reads from the seed files in content/
 *   - sanity-adapter.ts  fetches from Sanity (see lib/sanity/)
 *
 * Which one is active is decided below, automatically: Sanity is used once
 * NEXT_PUBLIC_SANITY_PROJECT_ID is set, otherwise the static seed data is
 * used so the site keeps working with zero configuration.
 *
 * To add a different CMS later (Strapi, Contentful, Payload...), implement
 * the same six functions in a new adapter file and select it below. No page
 * or component needs to change.
 */

export type { Product, ProductCategory } from "@content/products";
export type { Industry, CompanyValue, GalleryItem } from "@content/content";

import * as staticAdapter from "./static-adapter";

const useSanity = Boolean(process.env.NEXT_PUBLIC_SANITY_PROJECT_ID);

// Sanity's client throws at import time if env vars are missing, so it's
// only imported when actually configured. This keeps the static fallback
// working out of the box for anyone who clones the repo without Sanity set up.
const adapter = useSanity
  ? await import("./sanity-adapter")
  : staticAdapter;

export const getProducts = adapter.getProducts;
export const getFeaturedProducts = adapter.getFeaturedProducts;
export const getProductsByCategory = adapter.getProductsByCategory;
export const getProductCategories = adapter.getProductCategories;
export const getIndustries = adapter.getIndustries;
export const getGalleryItems = adapter.getGalleryItems;
export const getCompanyValues = adapter.getCompanyValues;
/**
 * Sanity adapter.
 *
 * Fetches content from Sanity, reshaped to match the same types the static
 * adapter returns, so components never need to know which one is active.
 */

import { sanityClient } from "@/lib/sanity/client";
import {
  productsQuery,
  featuredProductsQuery,
  productsByCategoryQuery,
  industriesQuery,
  galleryItemsQuery,
  companyValuesQuery,
} from "@/lib/sanity/queries";
import type { Product, ProductCategory } from "@content/products";
import type { Industry, CompanyValue, GalleryItem } from "@content/content";

const PRODUCT_CATEGORIES: ProductCategory[] = [
  "Tools & Hardware",
  "Fabrication",
  "PPE",
  "Stationery",
  "Ice Blocks",
];

export async function getProducts(): Promise<Product[]> {
  return sanityClient.fetch(productsQuery);
}

export async function getFeaturedProducts(): Promise<Product[]> {
  return sanityClient.fetch(featuredProductsQuery);
}

export async function getProductsByCategory(
  category: string,
): Promise<Product[]> {
  if (category === "All") return getProducts();
  return sanityClient.fetch(productsByCategoryQuery, { category });
}

export async function getProductCategories(): Promise<ProductCategory[]> {
  // Categories are a fixed enum in the schema (see sanity/schemaTypes/product.ts),
  // not their own document type, so this is returned directly rather than queried.
  return PRODUCT_CATEGORIES;
}

export async function getIndustries(): Promise<Industry[]> {
  return sanityClient.fetch(industriesQuery);
}

export async function getGalleryItems(): Promise<GalleryItem[]> {
  return sanityClient.fetch(galleryItemsQuery);
}

export async function getCompanyValues(): Promise<CompanyValue[]> {
  return sanityClient.fetch(companyValuesQuery);
}
/**
 * Static-seed adapter.
 *
 * Reads content straight from the seed files in `content/`. This is the
 * fallback used automatically if Sanity isn't configured (no
 * NEXT_PUBLIC_SANITY_PROJECT_ID set), and can also be selected explicitly.
 */

import { seedProducts, productCategories } from "@content/products";
import {
  industries as seedIndustries,
  galleryItems as seedGalleryItems,
  companyValues as seedCompanyValues,
} from "@content/content";
import type { Product, ProductCategory } from "@content/products";
import type { Industry, CompanyValue, GalleryItem } from "@content/content";

export async function getProducts(): Promise<Product[]> {
  return seedProducts;
}

export async function getFeaturedProducts(): Promise<Product[]> {
  return seedProducts.filter((p) => p.featured);
}

export async function getProductsByCategory(
  category: string,
): Promise<Product[]> {
  if (category === "All") return seedProducts;
  return seedProducts.filter((p) => p.category === category);
}

export async function getProductCategories(): Promise<ProductCategory[]> {
  return productCategories;
}

export async function getIndustries(): Promise<Industry[]> {
  return seedIndustries;
}

export async function getGalleryItems(): Promise<GalleryItem[]> {
  return seedGalleryItems;
}

export async function getCompanyValues(): Promise<CompanyValue[]> {
  return seedCompanyValues;
}

export const quoteWhatsAppHref = (productName?: string) => {
  const text = productName
    ? `Hello Canbri, I'd like to request a quote for ${productName}.`
    : "Hello Canbri, I'd like to request a quote.";
  return `https://wa.me/${siteConfig.whatsappNumber.replace(/[^\d]/g, "")}?text=${encodeURIComponent(
    text,
  )}`;
};
